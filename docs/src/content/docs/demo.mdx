<script>
{`
(async () => {
  if (!document.querySelector('link[href*="xterm"]')) {
    const l = document.createElement('link');
    l.rel = 'stylesheet';
    l.href = 'https://cdn.jsdelivr.net/npm/@xterm/xterm@5.5.0/css/xterm.min.css';
    document.head.appendChild(l);
    await new Promise(r => { l.onload = r; l.onerror = r; });
  }

  const { Terminal } = await import('https://cdn.jsdelivr.net/npm/@xterm/xterm@5.5.0/+esm');
  const { FitAddon } = await import('https://cdn.jsdelivr.net/npm/@xterm/addon-fit@0.10.0/+esm');
  const { Unicode11Addon } = await import('https://cdn.jsdelivr.net/npm/@xterm/addon-unicode11@0.8.0/+esm');

  const tv = document.getElementById('tv');
  const demoRoot = document.getElementById('demo-root');
  const demoStage = document.getElementById('demo-stage');
  const statusEl = document.getElementById('status');

  const term = new Terminal({
    cols: 80,
    rows: 24,
    cursorBlink: true,
    cursorStyle: 'block',
    fontSize: 14,
    lineHeight: 1.10,
    fontFamily: '"JetBrains Mono", "Cascadia Code", "Fira Code", Menlo, Monaco, monospace',
    theme: {
      background: '#121218',
      foreground: '#e0e0e0',
      cursor: '#00c8b4',
      selectionBackground: '#00644080',
      black: '#121218',
      red: '#ff5555',
      green: '#00c8b4',
      yellow: '#f1fa8c',
      blue: '#6c8fff',
      magenta: '#ff79c6',
      cyan: '#00ffc8',
      white: '#e0e0e0',
      brightBlack: '#3c3c50',
      brightGreen: '#00ffc8',
      brightCyan: '#00ffc8',
      brightWhite: '#ffffff',
    },
    allowProposedApi: true,
  });

  const fitAddon = new FitAddon();
  term.loadAddon(fitAddon);

  const u11 = new Unicode11Addon();
  term.loadAddon(u11);
  term.unicode.activeVersion = '11';

  function syncTerminalMetrics() {
    const hostWidth = demoStage.clientWidth || demoRoot.clientWidth || window.innerWidth;
    const nextFontSize =
      hostWidth <= 420 ? 11 :
      hostWidth <= 720 ? 12 :
      hostWidth <= 960 ? 13 :
      14;
    const nextLineHeight = hostWidth <= 720 ? 1.05 : 1.10;
    let changed = false;

    if (term.options.fontSize !== nextFontSize) {
      term.options.fontSize = nextFontSize;
      changed = true;
    }

    if (term.options.lineHeight !== nextLineHeight) {
      term.options.lineHeight = nextLineHeight;
      changed = true;
    }

    return changed;
  }

  let fitRaf = 0;
  const scheduleFit = () => {
    if (fitRaf) cancelAnimationFrame(fitRaf);
    fitRaf = requestAnimationFrame(() => {
      fitRaf = 0;
      const metricsChanged = syncTerminalMetrics();
      fitAddon.fit();
      if (metricsChanged && term.rows > 0) {
        term.refresh(0, term.rows - 1);
      }
    });
  };

  term.open(tv);
  scheduleFit();
  setTimeout(scheduleFit, 50);
  setTimeout(scheduleFit, 200);

  const isProduction = window.location.hostname === 'dotsider.dev';
  const wsProto = isProduction ? 'wss' : 'ws';
  const wsHost = isProduction ? 'dotsider.dev' : 'localhost:64219';
  const wsUrl = wsProto + '://' + wsHost + '/ws';
  let ws;
  let wasConnected = false;
  let lastConnectTime = 0;

  function sendResize() {
    if (ws?.readyState !== WebSocket.OPEN) return;
    ws.send(JSON.stringify({
      type: 'resize',
      cols: term.cols,
      rows: term.rows
    }));
  }

  function connect() {
    wasConnected = false;
    lastConnectTime = Date.now();
    statusEl.textContent = 'Connecting...';
    statusEl.style.color = '#f1fa8c';
    ws = new WebSocket(wsUrl);

    const timeout = setTimeout(() => { ws.close(); }, 5000);

    ws.onopen = () => {
      clearTimeout(timeout);
      wasConnected = true;
      statusEl.textContent = 'Connected';
      statusEl.style.color = '#00c8b4';
      scheduleFit();
      setTimeout(sendResize, 0);
    };
    ws.onmessage = (e) => {
      if (e.data.includes('\\x1b[c') && ws.readyState === WebSocket.OPEN) ws.send('\\x1b[?62;4c');
      term.write(e.data);
    };
    ws.onclose = () => {
      clearTimeout(timeout);
      // Only auto-restart if session lasted more than 3s (not a rapid cycle)
      if (wasConnected && Date.now() - lastConnectTime > 3000) {
        statusEl.textContent = 'Restarting...';
        statusEl.style.color = '#f1fa8c';
        term.reset();
        setTimeout(connect, 1500);
      } else if (!wasConnected) {
        statusEl.textContent = 'Offline';
        statusEl.style.color = '#ff5555';
      } else {
        statusEl.textContent = 'Offline';
        statusEl.style.color = '#ff5555';
      }
    };
    ws.onerror = () => { clearTimeout(timeout); };
  }

  term.onData((d) => { if (ws?.readyState === WebSocket.OPEN) ws.send(d); });
  term.onBinary((d) => { if (ws?.readyState === WebSocket.OPEN) ws.send(d); });
  term.onResize(sendResize);
  window.addEventListener('resize', scheduleFit);
  window.visualViewport?.addEventListener('resize', scheduleFit);
  new ResizeObserver(scheduleFit).observe(demoRoot);

  document.getElementById('reconnect-btn').addEventListener('click', () => {
    if (ws) { ws.onclose = null; ws.close(); }
    statusEl.textContent = 'Reconnecting...';
    statusEl.style.color = '#f1fa8c';
    setTimeout(connect, 1500);
  });

  connect();
})();
`}
</script>
